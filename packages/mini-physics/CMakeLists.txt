cmake_minimum_required(VERSION 3.20)
project(seedcore_mini_physics LANGUAGES CXX)
add_library(mini_physics STATIC src/mini_physics.cpp)
target_include_directories(mini_physics PUBLIC include)
target_compile_features(mini_physics PUBLIC cxx_std_17)
target_compile_options(mini_physics PRIVATE -ffp-contract=off -fno-fast-math)
add_executable(mini_physics_cli src/native_cli.cpp)
target_link_libraries(mini_physics_cli PRIVATE mini_physics)

enable_testing()
add_executable(mini_physics_tests src/native_tests.cpp)
target_link_libraries(mini_physics_tests PRIVATE mini_physics)
target_compile_options(mini_physics_tests PRIVATE -UNDEBUG)
add_test(NAME native_invariants COMMAND mini_physics_tests)
